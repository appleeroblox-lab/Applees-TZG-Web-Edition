const origin = location.origin;
const threePath = new URL('/vendor/three.module.js', origin);
const loaderPath = new URL('/models.js', origin);

export async function mount(root, api) {
  const [T, { modelFromData }, draft] = await Promise.all([
    import(threePath), import(loaderPath), api.call('read'),
  ]);
  if (draft.status !== 'ok') throw new Error('Could not load the draft.');
  async function loadModel(id) {
    const response = await fetch('/models/' + encodeURIComponent(id) + '.json');
    if (!response.ok) throw new Error('Could not load the selected model.');
    const data = await response.json();
    for (const mesh of data.meshes) {
      for (const field of ['map', 'emissiveMap']) {
        if (mesh[field]) mesh[field] = new URL(mesh[field], origin).href;
      }
    }
    return data;
  }
  const shipId = 'player-ship';
  let modelId = shipId;
  let model = await loadModel(modelId);
  let revision = draft.revision;
  function getMaterials(document) {
    return { [shipId]: document.shipMaterials ?? [], ...document.modelMaterials };
  }
  let materials = getMaterials(draft.document);
  let edits = materials[modelId] ?? [];
  const undo = [];
  const redo = [];

  root.innerHTML = `
    <header class="studioRibbon">
      
      <div class="tools" aria-label="Editing toolbar">
        <details class="studioFile"><summary>File</summary><div>

          <button data-tool="select" aria-pressed="true">Select</button>
          <button data-tool="position" aria-pressed="false">Move</button>

          <button data-tool="scale" aria-pressed="false">Scale</button>
        </div>
        <button id="undo">Undo</button><button id="redo">Redo</button>
        <button id="zoom-in">Zoom In</button><button id="zoom-out">Zoom Out</button>
        <button id="focus">Focus Selection</button><button id="grid" aria-pressed="false">Grid</button>
        <button id="history">Version History</button><button id="lock">Lock Studio</button>
      </div>

    <div class="panels">
      <section class="assetsPane" data-panel="assets">
        <label>Search models<input id="asset-search" type="search"></label>
        <label>Model<select id="asset" size="12"></select></label>

      <section class="viewport" aria-label="Live model preview">

          <label>Part<select id="part" size="5"></select></label></section>

          <label class="studioVisibility"><span>Visible</span><input id="part-visible" type="checkbox" checked></label>
          <div id="transform-fields"></div>
          <label>Colour<button type="button" id="model-colour" aria-label="Colour"></button></label>
          <label>Metalness<input id="metalness" type="range" min="0" max="1" step=".01"></label>
          <label>Roughness<input id="roughness" type="range" min="0" max="1" step=".01"></label>
          <label>Opacity<input id="opacity" type="range" min="0" max="1" step=".01"></label>
        </section>

    <dialog id="save-menu" aria-labelledby="save-title">
      <h2 id="save-title">Save Draft</h2>

        <p id="save-status" role="status"></p>
      </form>
    </dialog>
    <dialog id="update-info" aria-labelledby="update-title">
      <h2 id="update-title">Update Info</h2>
      <form id="update-form">
      <label>Update title<input id="update-name" maxlength="80"></label>
      <label>Update notes<textarea id="update-notes" maxlength="1200"></textarea></label>
      <p id="update-summary"></p>
      <p>Publishing to the live game is unavailable in this preview.</p>
      <label>Studio password<input id="update-secret" type="password" minlength="12" maxlength="72" autocomplete="off" required></label>
      <div class="tools"><button id="keep-info">Save Update Info</button><button type="button" id="close-info">Close</button></div>
      <p id="publish-status" role="status"></p>
      </form>
    </dialog>

      <p id="studio-output" role="status">Model loaded.</p>
      <div id="versions" hidden></div>
    </section>`;
  const message = text => { root.querySelector('#studio-output').textContent = text; };
  const find = selector => root.querySelector(selector);
  const viewport = find('.viewport');
  const layout = workspaceLayout(root, viewport, find('#view-panels'));
  for (const [id, title] of Object.entries({ assets: 'Assets', explorer: 'Explorer', properties: 'Properties', output: 'Output' })) {
    layout.register(id, title, find('[data-panel="' + id + '"]'));
  }
  find('.inspector').remove();
  const renderer = new T.WebGLRenderer({ antialias: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  viewport.append(renderer.domElement);
  const axisOverlay = document.createElement('div');
  axisOverlay.className = 'studioAxisOverlay';
  const axisMarkers = ['X', 'Y', 'Z'].map((axis, index) => {
    const marker = document.createElement('span');
    marker.textContent = axis;
    marker.dataset.axis = axis;
    marker.style.color = ['#ff6e79', '#71ed9e', '#73b6ff'][index];
    axisOverlay.append(marker);
    return marker;
  });
  axisOverlay.hidden = true;
  viewport.append(axisOverlay);
  renderer.domElement.tabIndex = 0;
  renderer.domElement.setAttribute('aria-label', 'Model viewport');
  const scene = new T.Scene();
  scene.background = new T.Color(0x102544);
  const camera = new T.PerspectiveCamera(45, 1, 0.01, 100000);
  scene.add(new T.HemisphereLight(0xffffff, 0x526e99, 2.3));
  const key = new T.DirectionalLight(0xffffff, 2.1);
  key.position.set(2, 4, 3);
  scene.add(key);
  const center = new T.Vector3();
  let radius = 1;
  let azimuth = .65;
  let elevation = .42;
  let distance = 1;
  let ship;
  let selectionOutline;
  let grid;
  let gizmo;
  let tool = 'select';
  const axisDirections = [new T.Vector3(1, 0, 0), new T.Vector3(0, 1, 0), new T.Vector3(0, 0, 1)];
  const axisColors = [0xff6e79, 0x71ed9e, 0x73b6ff];
  const transformDefaults = { position: [0, 0, 0], rotation: [0, 0, 0], scale: [1, 1, 1] };
  const axisLabels = ['X', 'Y', 'Z'];
  let gridVisible = false;
  const raycaster = new T.Raycaster();
  const rayPoint = new T.Vector2();
  let selected = 0;
  const partList = find('#part');
  const fields = ['color', 'metalness', 'roughness', 'opacity'];

  function draw() {
    const w = Math.max(viewport.clientWidth, 1);
    const h = Math.max(viewport.clientHeight - find('#scene-title').offsetHeight, 1);
    if (!viewport.clientWidth || !viewport.clientHeight || renderer.domElement.hidden) return;
    renderer.setSize(w, h);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    const offset = new T.Vector3().setFromSphericalCoords(distance * radius, Math.PI / 2 - elevation, azimuth);
    camera.position.copy(center).add(offset);
    camera.lookAt(center);
    renderer.render(scene, camera);
    axisOverlay.hidden = !gizmo;
    if (gizmo) for (const point of gizmo.children) {
      const axis = point.userData.axis;
      if (!Number.isInteger(axis)) continue;
      const projected = (point.userData.label ? gizmo.localToWorld(point.userData.label.clone()) : point.getWorldPosition(new T.Vector3())).project(camera);
      const marker = axisMarkers[axis];
      marker.style.left = ((projected.x + 1) * w / 2) + 'px';
      marker.style.top = ((1 - projected.y) * h / 2 + find('#scene-title').offsetHeight) + 'px';
    }
  }
  function release(group) {
    if (!group) return;
    scene.remove(group);
    group.traverse(child => {
      child.geometry?.dispose();
      child.material?.dispose();
    });
  }
  function partValue(field, index = selected) {
    return edits.find(edit => edit.index === index && edit.field === field)?.value ??
      (field === 'visible' ? true : transformDefaults[field]);
  }
  function applyTransform(mesh, values) {
    mesh.geometry.computeBoundingBox();
    const pivot = mesh.geometry.boundingBox.getCenter(new T.Vector3());
    mesh.scale.fromArray(values.scale);
    mesh.rotation.set(...values.rotation.map(angle => T.MathUtils.degToRad(angle)));
    mesh.position.fromArray(values.position).add(pivot).sub(
      pivot.clone().multiply(mesh.scale).applyQuaternion(mesh.quaternion)
    );
    mesh.visible = values.visible;
    mesh.updateMatrixWorld(true);
  }
  function rebuildGizmo() {
    release(gizmo);
    gizmo = undefined;
    if (tool === 'select' || !ship.children[selected].visible) return;
    gizmo = new T.Group();
    const part = ship.children[selected];
    new T.Box3().setFromObject(part).getCenter(gizmo.position);
    const length = Math.max(radius * .48, .003);
    for (const [index, axis] of axisDirections.entries()) {
      const material = new T.MeshBasicMaterial({ color: axisColors[index], depthTest: false, depthWrite: false });
      let endpoint;
      if (tool === 'rotation') {
        endpoint = new T.Mesh(new T.TorusGeometry(length, length * .035, 8, 48), material);
        endpoint.quaternion.setFromUnitVectors(axisDirections[2], axis);
        endpoint.userData.label = axisDirections[(index + 1) % 3].clone().multiplyScalar(length);
      } else {
        const line = new T.Line(new T.BufferGeometry().setFromPoints([
          new T.Vector3(), axis.clone().multiplyScalar(length),
        ]), new T.LineBasicMaterial({ color: axisColors[index], depthTest: false, depthWrite: false }));
        line.renderOrder = 5;
        gizmo.add(line);
        endpoint = new T.Mesh(tool === 'scale' ? new T.BoxGeometry(length * .2, length * .2, length * .2) :
          new T.ConeGeometry(length * .11, length * .23, 12), material);
        if (tool === 'position') endpoint.quaternion.setFromUnitVectors(axisDirections[1], axis);
        endpoint.position.copy(axis).multiplyScalar(length);
      }
      endpoint.userData.axis = index;
      endpoint.renderOrder = 6;
      gizmo.add(endpoint);
    }
    scene.add(gizmo);
  }
  function commitProperty(field, value) {
    undo.push(edits);
    redo.length = 0;
    const edit = { index: selected, name: model.meshes[selected].name, field, value };
    edits = [...edits.filter(item => item.index !== selected || item.field !== field), edit];
    update();
  }
  const transformInputs = new Map();
  for (const [field, title] of Object.entries({ position: 'Position', rotation: 'Rotation', scale: 'Scale' })) {
    const row = document.createElement('fieldset');
    const label = document.createElement('legend');
    label.textContent = title;
    row.append(label);
    const inputs = axisLabels.map((axis, index) => {
      const fieldLabel = document.createElement('label');
      fieldLabel.textContent = axis;
      const input = document.createElement('input');
      input.type = 'number';
      input.step = 'any';
      input.min = field === 'scale' ? '.01' : '-10000';
      input.max = field === 'scale' ? '100' : '10000';
      input.setAttribute('aria-label', title + ' ' + axis);
      input.onchange = () => {
        if (!input.value || !input.checkValidity()) return update();
        const value = [...partValue(field)];
        value[index] = Number(input.value);
        commitProperty(field, value);
      };
      fieldLabel.append(input);
      row.append(fieldLabel);
      return input;
    });
    transformInputs.set(field, inputs);
    find('#transform-fields').append(row);
  }
  find('#part-visible').onchange = event => commitProperty('visible', event.target.checked);
  for (const button of root.querySelectorAll('[data-tool]')) {
    button.onclick = () => {
      tool = button.dataset.tool;
      for (const item of root.querySelectorAll('[data-tool]')) item.setAttribute('aria-pressed', String(item === button));
      rebuildGizmo();
      draw();
    };
  }
  function update(resetCamera = false) {
    const data = applyModelMaterialEdits(model, edits);
    release(ship);
    if (selectionOutline) release(selectionOutline);
    ship = modelFromData(data);
    scene.add(ship);
    for (const [index, mesh] of ship.children.entries()) {
      applyTransform(mesh, {
        position: partValue('position', index), rotation: partValue('rotation', index),
        scale: partValue('scale', index), visible: partValue('visible', index),
      });
    }
    if (resetCamera) {
      const sphere = new T.Box3().setFromObject(ship).getBoundingSphere(new T.Sphere());
      radius = Math.max(sphere.radius, .01);
      center.copy(sphere.center);
      distance = 2.8;
    }
    const item = data.meshes[selected];
    selectionOutline = new T.BoxHelper(ship.children[selected], 0x83d8ff);
    selectionOutline.visible = ship.children[selected].visible;
    scene.add(selectionOutline);
    if (resetCamera) {
      release(grid);
      grid = new T.GridHelper(radius * 5, 20, 0x5183a5, 0x2a4b64);
      grid.position.copy(center);
      grid.position.y -= radius * .55;
      grid.visible = gridVisible;
      scene.add(grid);
    }
    rebuildGizmo();
    find('#part-visible').checked = partValue('visible');
    for (const [field, inputs] of transformInputs) {
      const value = partValue(field);
      for (const [index, input] of inputs.entries()) input.value = Number(value[index].toFixed(4));
    }
    for (const field of fields) {
      const value = item[field];
      if (field === 'color') {
        const color = '#' + new T.Color(value).getHexString();
        const swatch = find('#model-colour');
        swatch.style.setProperty('--swatch', color);
        swatch.dataset.color = color;
        swatch.textContent = color.toUpperCase();
        continue;
      }
      find('#' + field).value = value ?? (field === 'metalness' ? 0 : 1);
    }
    draw();
  }
  function parts() {
    const search = find('#search').value.trim().toLowerCase();
    const matches = model.meshes.flatMap((item, index) => item.name.toLowerCase().includes(search)
      ? [new Option(item.name, String(index), false, index === selected)] : []);
    partList.replaceChildren(...matches);
  }
  const assetPicker = find('#asset');
  const assetLabel = id => id === shipId ? 'SolarFlare' : id;
  function filterAssets() {
    const search = find('#asset-search').value.trim().toLowerCase();
    assetPicker.replaceChildren(...modelCatalog
      .filter(id => assetLabel(id).toLowerCase().includes(search))
      .map(id => new Option(assetLabel(id), id, false, id === modelId)));
    assetPicker.value = modelId;
  }
  filterAssets();
  find('#asset-search').oninput = filterAssets;
  assetPicker.onchange = async () => {
    const nextId = assetPicker.value;
    assetPicker.disabled = true;
    try {
      const next = await loadModel(nextId);
      applyModelMaterialEdits(next, materials[nextId] ?? []);
      materials[modelId] = edits;
      modelId = nextId;
      model = next;
      find('#scene-title').textContent = assetLabel(nextId);
      message('Loaded ' + assetLabel(nextId) + '.');
      selected = 0;
      edits = materials[modelId] ?? [];
      undo.length = redo.length = 0;
      parts();
      update(true);
    } catch (error) {
      assetPicker.value = modelId;
      message(error.message);
    } finally { assetPicker.disabled = false; }
  };
  find('#search').addEventListener('input', parts);
  partList.addEventListener('change', () => {
    selected = Number(partList.value);
    update();
  });
  for (const field of fields.filter(name => name !== 'color')) {
    find('#' + field).addEventListener('change', event => {
      const value = field === 'color' ? event.target.value : Number(event.target.value);
      commitProperty(field, value);
    });
  }
  find('#model-colour').onclick = () => openPaintPanel(root, {
    layout, color: find('#model-colour').dataset.color,
    onDone: color => commitProperty('color', color),
  });
  find('#undo').onclick = () => {
    if (closeUi) return closeUi.undo();
    if (!undo.length) return;
    redo.push(edits);
    edits = undo.pop();
    update();
  };
  find('#redo').onclick = () => {
    if (closeUi) return closeUi.redo();
    if (!redo.length) return;
    undo.push(edits);
    edits = redo.pop();
    update();
  };
  find('#lock').onclick = api.lock;
  let pointer;
  let axisDrag;
  const touches = new Map();
  let pinch;
  function touchPair() {
    if (touches.size < 2) return;
    const [a, b] = touches.values();
    return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2, gap: Math.max(1, Math.hypot(a.x - b.x, a.y - b.y)) };
  }
  function panCamera(dx, dy) {
    const speed = distance * radius / Math.max(100, renderer.domElement.clientHeight);
    center.addScaledVector(new T.Vector3().setFromMatrixColumn(camera.matrixWorld, 0), -dx * speed);
    center.addScaledVector(new T.Vector3().setFromMatrixColumn(camera.matrixWorld, 1), dy * speed);
  }
  function endTouch(event) {
    touches.delete(event.pointerId);
    if (pinch) {
      if (!touches.size) pinch = undefined;
      else pinch = touchPair() ?? pinch;
      return true;
    }
    return false;
  }
  function pointRay(event) {
    const rect = renderer.domElement.getBoundingClientRect();
    rayPoint.set((event.clientX - rect.left) / rect.width * 2 - 1,
      -(event.clientY - rect.top) / rect.height * 2 + 1);
    raycaster.setFromCamera(rayPoint, camera);
  }
  function beginAxis(event) {
    if (!gizmo || event.button !== 0) return;
    pointRay(event);
    const hit = raycaster.intersectObjects(gizmo.children.filter(child => Number.isInteger(child.userData.axis)), false)[0];
    if (!hit) return;
    const axis = hit.object.userData.axis;
    const start = gizmo.position.clone().project(camera);
    const directionWorld = tool === 'rotation'
      ? axisDirections[axis].clone().cross(hit.point.clone().sub(gizmo.position)).normalize()
      : axisDirections[axis];
    const end = gizmo.position.clone().add(directionWorld.clone().multiplyScalar(radius)).project(camera);
    const rect = renderer.domElement.getBoundingClientRect();
    const direction = new T.Vector2((end.x - start.x) * rect.width / 2, -(end.y - start.y) * rect.height / 2);
    const pixels = Math.max(20, direction.length());
    direction.normalize();
    if (!direction.lengthSq()) direction.set(1, 0);
    axisDrag = { id: event.pointerId, axis, direction, pixels, x: event.clientX, y: event.clientY,
      field: tool, initial: [...partValue(tool)] };
    renderer.domElement.setPointerCapture(event.pointerId);
  }
  renderer.domElement.oncontextmenu = event => event.preventDefault();
  renderer.domElement.onpointerdown = event => {
    if (event.pointerType === 'touch') {
      touches.set(event.pointerId, { x: event.clientX, y: event.clientY });
      if (touches.size > 1) {
        event.preventDefault();
        renderer.domElement.setPointerCapture(event.pointerId);
        pointer = undefined;
        if (axisDrag) { axisDrag = undefined; update(); }
        pinch = touchPair();
        return;
      }
    }
    if (pointer || axisDrag || ![0, 1, 2].includes(event.button)) return;
    event.preventDefault();
    renderer.domElement.focus({ preventScroll: true });
    beginAxis(event);
    if (axisDrag) return;
    pointer = { id: event.pointerId, x: event.clientX, y: event.clientY, moved: false,
      pan: event.button > 0 || event.shiftKey };
    renderer.domElement.setPointerCapture(event.pointerId);
  };
  renderer.domElement.onpointermove = event => {
    if (touches.has(event.pointerId)) {
      touches.set(event.pointerId, { x: event.clientX, y: event.clientY });
      if (pinch) {
        const next = touchPair();
        if (next) {
          panCamera(next.x - pinch.x, next.y - pinch.y);
          zoom(pinch.gap / next.gap);
          pinch = next;
        }
        return;
      }
    }
    if (axisDrag?.id === event.pointerId) {
      const { x, y, axis, direction, pixels, field, initial } = axisDrag;
      const delta = ((event.clientX - x) * direction.x + (event.clientY - y) * direction.y) / pixels;
      const value = [...initial];
      const factor = field === 'rotation' ? 90 : field === 'position' ? radius : 1;
      value[axis] = field === 'scale' ? Math.min(100, Math.max(.01, initial[axis] * (1 + delta))) :
        Math.min(10000, Math.max(-10000, initial[axis] + delta * factor));
      axisDrag.value = value;
      applyTransform(ship.children[selected], {
        position: partValue('position'), rotation: partValue('rotation'), scale: partValue('scale'), visible: true,
        [field]: value,
      });
      selectionOutline.update();
      draw();
      return;
    }
    if (pointer?.id !== event.pointerId) return;
    const dx = event.clientX - pointer.x;
    const dy = event.clientY - pointer.y;
    if (!pointer.moved && Math.hypot(dx, dy) < 4) return;
    pointer.moved = true;
    if (pointer.pan) {
      panCamera(dx, dy);
    } else {
      azimuth -= dx * .012;
      elevation = Math.max(-1.45, Math.min(1.45, elevation + dy * .012));
    }
    pointer.x = event.clientX;
    pointer.y = event.clientY;
    draw();
  };
  renderer.domElement.onpointerup = event => {
    if (endTouch(event)) return;
    if (axisDrag?.id === event.pointerId) {
      const { field, value } = axisDrag;
      axisDrag = undefined;
      if (value) commitProperty(field, value);
      return;
    }
    if (pointer?.id !== event.pointerId) return;
    const select = !pointer.moved && !pointer.pan;
    pointer = undefined;
    if (!select) return;
    pointRay(event);
    const hit = raycaster.intersectObjects(ship.children.filter(mesh => mesh.visible), false)[0];
    if (!hit) return;
    selected = ship.children.indexOf(hit.object);
    parts();
    update();
  };
  renderer.domElement.onpointercancel = renderer.domElement.onlostpointercapture = event => {
    endTouch(event);
    if (pointer?.id === event.pointerId) pointer = undefined;
    if (axisDrag?.id === event.pointerId) { axisDrag = undefined; update(); }
  };
  find('#focus').onclick = () => {
    new T.Box3().setFromObject(ship.children[selected]).getCenter(center);
    draw();
  };
  renderer.domElement.onkeydown = event => {
    const keys = { q: 'select', w: 'position', e: 'rotation', r: 'scale' };
    const key = event.key.toLowerCase();
    if (keys[key]) find('[data-tool="' + keys[key] + '"]').click();
    else if (key === 'f') find('#focus').click();
    else return;
    event.preventDefault();
  };
  find('#grid').onclick = () => {
    gridVisible = !gridVisible;
    grid.visible = gridVisible;
    find('#grid').setAttribute('aria-pressed', String(gridVisible));
    draw();
  };
  function zoom(factor) {
    distance = Math.max(1.1, Math.min(12, distance * factor));
    draw();
  }
  function zoomCurrent(factor) {
    if (closeUi) closeUi.zoom(factor);
    else if (scriptStudio) scriptStudio.zoom(factor);
    else zoom(factor);
  }
  find('#zoom-in').onclick = () => zoomCurrent(.8);
  find('#zoom-out').onclick = () => zoomCurrent(1.25);
  renderer.domElement.addEventListener('wheel', event => {
    event.preventDefault();
    zoom(Math.exp(event.deltaY * .001));
  }, { passive: false });
  const draftForm = find('#draft');
  const secret = draftForm.querySelector('[type=password]');
  const saveMenu = find('#save-menu');
  let saving = false;
  let restoreRevision = null;
  let closeUi;
  let scriptStudio;
  let activeStudio = 'model';
  let switching = false;
  let uiEdits = draft.document.ui ?? {};
  let updateInfo = draft.document.updateInfo ?? {};
  const studios = { model: 'Model Studio', ui: 'UI Studio', script: 'Script Studio' };
  const modelPanels = ['assets', 'explorer', 'properties'];
  const studioSwitch = find('#switch-studio');
  async function switchStudio(next, reload = false) {
    if (switching || (next === activeStudio && !reload)) return;
    switching = true;
    layout.close('colour');
    layout.close('gradient');
    closeUi?.();
    closeUi = undefined;
    scriptStudio?.close();
    scriptStudio = undefined;
    renderer.domElement.hidden = next !== 'model';
    axisOverlay.hidden = true;
    layout.suspend(next === 'model' ? [] : modelPanels);
    try {
      if (next === 'ui') closeUi = await mountInterfaceEditor(root, viewport, uiEdits, edits => { uiEdits = edits; }, layout);
      if (next === 'script') scriptStudio = await mountScriptStudio(viewport, layout);
      activeStudio = next;
    } catch (error) {
      activeStudio = 'model';
      renderer.domElement.hidden = false;
      layout.suspend([]);
      message(error.message);
    } finally {
      switching = false;
      const modelMode = activeStudio === 'model';
      find('#active-studio').textContent = studios[activeStudio];
      find('#scene-title').textContent = modelMode ? assetLabel(modelId) : activeStudio === 'ui' ? 'UI Preview' : 'Read-only game scripts';
      viewport.setAttribute('aria-label', modelMode ? 'Live model preview' : studios[activeStudio]);
      for (const id of ['#model-tools', '#focus', '#grid']) find(id).hidden = !modelMode;
      for (const id of ['#undo', '#redo']) find(id).hidden = activeStudio === 'script';
      for (const button of studioSwitch.querySelectorAll('[data-studio]')) {
        button.setAttribute('aria-pressed', String(button.dataset.studio === activeStudio));
      }
      if (modelMode) draw();
    }
  }
  for (const button of studioSwitch.querySelectorAll('[data-studio]')) {
    button.setAttribute('aria-pressed', String(button.dataset.studio === activeStudio));
    button.onclick = () => { studioSwitch.open = false; void switchStudio(button.dataset.studio); };
  }
  for (const details of root.querySelectorAll('.studioFile')) {
    details.addEventListener('toggle', () => {
      if (!details.open) return;
      for (const other of root.querySelectorAll('.studioFile[open]')) if (other !== details) other.open = false;
      const rect = details.querySelector('summary').getBoundingClientRect();
      const popup = details.querySelector('div');
      popup.style.left = Math.max(4, Math.min(rect.left, innerWidth - popup.offsetWidth - 4)) + 'px';
      popup.style.top = rect.bottom + 'px';
    });
  }
  find('.studioRibbon > .tools').addEventListener('scroll', () => {
    for (const menu of root.querySelectorAll('.studioFile[open]')) menu.open = false;
  });
  const fileMenu = find('.studioFile');
  function openSave(version = null) {
    restoreRevision = version;
    const title = version === null ? 'Save Draft' : 'Restore Draft';
    find('#save-title').textContent = title;
    find('#save-confirm').textContent = title;
    find('#save-status').textContent = '';
    secret.value = '';
    saveMenu.showModal();
    secret.focus();
  }
  find('#file-save').onclick = () => { fileMenu.open = false; openSave(); };
  find('#save-cancel').onclick = () => saveMenu.close();
  saveMenu.addEventListener('close', () => { secret.value = ''; });
  const info = find('#update-info');
  const publishSecret = find('#update-secret');
  find('#file-update').onclick = () => {
    fileMenu.open = false;
    find('#update-name').value = updateInfo.title ?? '';
    find('#update-notes').value = updateInfo.notes ?? '';
    publishSecret.value = '';
    find('#publish-status').textContent = '';
    find('#update-summary').textContent = 'UI menus changed: ' + Object.keys(uiEdits).length +
      ' · Models with changes: ' + Object.values({ ...materials, [modelId]: edits }).filter(items => items.length).length;
    info.showModal();
  };
  find('#close-info').onclick = () => info.close();
  info.addEventListener('close', () => { publishSecret.value = ''; });
  function draftPayload(metadata = updateInfo) {
    materials[modelId] = edits;
    const { shipMaterials, ...otherDraftFields } = draft.document;
    return { edits: { ...otherDraftFields, modelMaterials: materials, ui: uiEdits, updateInfo: metadata } };
  }
  async function write(payload, password, status) {
    if (saving) return;
    saving = true;
    try {
      const result = await api.call('write', { ...payload, expected_revision: revision, password });
      if (result.status !== 'ok') throw new Error('Draft write failed (' + result.status + ').');
      revision = result.revision;
      message('Draft version ' + revision + ' saved.');
      return true;
    } catch (error) { status.textContent = error.message; }
    finally { saving = false; }
  }
  find('#update-form').onsubmit = async event => {
    event.preventDefault();
    const metadata = { title: find('#update-name').value.trim(), notes: find('#update-notes').value.trim() };
    const password = publishSecret.value;
    publishSecret.value = '';
    if (await write(draftPayload(metadata), password, find('#publish-status'))) {
      updateInfo = metadata;
      info.close();
    }
  };
  draftForm.onsubmit = async event => {
    event.preventDefault();
    const version = restoreRevision;
    const password = secret.value;
    secret.value = '';
    if (!await write(version === null ? draftPayload() : { restore_revision: version }, password, find('#save-status'))) return;
    saveMenu.close();
    if (version === null) return;
    try {
      const restored = await api.call('read');
      if (restored.status !== 'ok') throw new Error('Could not reload the restored draft.');
      draft.document = restored.document;
      materials = getMaterials(restored.document);
      uiEdits = restored.document.ui ?? {};
      updateInfo = restored.document.updateInfo ?? {};
      edits = materials[modelId] ?? [];
      undo.length = redo.length = 0;
      update();
      if (activeStudio === 'ui') await switchStudio('ui', true);
    } catch (error) { message(error.message); }
  };
  find('#history').onclick = async () => {
    try {
      const { versions } = await api.call('history');
      layout.activate('output');
      const panel = find('#versions');
      panel.hidden = false;
      panel.replaceChildren();
      for (const version of versions) {
        const row = document.createElement('p');
        row.textContent = 'Version ' + version.revision + ' · ' + version.action + ' ';
        const button = document.createElement('button');
        button.textContent = 'Restore Draft';
        button.onclick = () => openSave(version.revision);
        row.append(button);
        panel.append(row);
      }
      if (!versions.length) panel.textContent = 'No draft versions yet.';
    } catch (error) { message(error.message); }
  };

  const loadingManager = T.DefaultLoadingManager;
  const originalOnLoad = loadingManager.onLoad;
  function onLoad() {
    originalOnLoad?.();
    draw();
  }
  loadingManager.onLoad = onLoad;
  parts();
  update(true);
  const viewportObserver = new ResizeObserver(draw);
  viewportObserver.observe(viewport);
  window.addEventListener('resize', draw);
  return () => {
    window.removeEventListener('resize', draw);
    viewportObserver.disconnect();
    layout.dispose();
    if (loadingManager.onLoad === onLoad) loadingManager.onLoad = originalOnLoad;
    closeUi?.();
    scriptStudio?.close();
    release(ship);
    release(selectionOutline);
    release(grid);
    release(gizmo);
    renderer.dispose();
    renderer.forceContextLoss();
  };
}
